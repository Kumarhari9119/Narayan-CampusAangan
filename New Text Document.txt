// LOAD SAVED ROOMS

function loadRooms() {
  try {
    const saved = JSON.parse(
      localStorage.getItem("roomkhoj_rooms") || "[]"
    );
    return Array.isArray(saved) ? saved : [];
  } catch {
    return [];
  }
}

let rooms = [...sampleRooms, ...loadRooms()];

// DISPLAY ROOMS

function displayRooms(list) {

  const grid = document.getElementById("roomGrid");

  if (list.length === 0) {
    grid.innerHTML = `
      <div class="empty">
        <h3>No rooms found!</h3>
        <p>Try changing your search filters.</p>
      </div>
    `;
    return;
  }

  grid.innerHTML = list.map(room => `
    <article class="room-card">

      <div class="room-image">
        <span class="tag">${escapeHTML(room.suitable)}</span>
        🏠
      </div>

      <div class="room-info">

        <h3>${escapeHTML(room.title)}</h3>

        <div class="location">
          📍 ${escapeHTML(room.area)}
        </div>

        <div class="price">
          ₹${Number(room.rent).toLocaleString("en-IN")}
          <small>/ month</small>
        </div>

        <div class="details">
          <span>${escapeHTML(room.type)} Room</span>
          <span>${escapeHTML(room.furnishing || "Unfurnished")}</span>
        </div>

        <div class="card-actions">
          <button class="btn outline"
            onclick="showDetails('${room.id}')">
            View Details
          </button>

          <button class="btn primary"
            onclick="contactOwner('${room.id}')">
            Contact
          </button>
        </div>

      </div>
    </article>
  `).join("");
}

// BASIC HTML ESCAPING

function escapeHTML(value) {
  return String(value ?? "").replace(/[&<>"']/g, char => ({
    "&": "&amp;",
    "<": "&lt;",
    ">": "&gt;",
    '"': "&quot;",
    "'": "&#39;"
  })[char]);
}

// SEARCH FILTER

function searchRooms() {

  const area = document.getElementById("searchArea")
    .value.trim().toLowerCase();

  const rent = Number(
    document.getElementById("searchRent").value
  );

  const type = document.getElementById("searchType").value;

  const filtered = rooms.filter(room => {

    const areaMatch =
      !area || room.area.toLowerCase().includes(area);

    const rentMatch =
      !rent || Number(room.rent) <= rent;

    const typeMatch =
      !type || room.type === type;

    return areaMatch && rentMatch && typeMatch;
  });

  displayRooms(filtered);

  document.getElementById("rooms").scrollIntoView({
    behavior: "smooth"
  });
}

// RESET SEARCH

function resetSearch() {
  document.getElementById("searchArea").value = "";
  document.getElementById("searchRent").value = "";
  document.getElementById("searchType").value = "";
  displayRooms(rooms);
}

// POST NEW ROOM

document.getElementById("postForm")
  .addEventListener("submit", function(e) {

    e.preventDefault();

    const formData = new FormData(this);

    const room = {
      id: "user_" + Date.now(),
      title: formData.get("title").trim(),
      owner: formData.get("owner").trim(),
      area: formData.get("area").trim(),
      rent: Number(formData.get("rent")),
      type: formData.get("type"),
      suitable: formData.get("suitable"),
      furnishing: formData.get("furnishing"),
      deposit: Number(formData.get("deposit") || 0),
      description: formData.get("description").trim(),
      phone: formData.get("phone").trim()
    };

    if (!Number.isFinite(room.rent) || room.rent <= 0) {
      alert("Please enter a valid rent.");
      return;
    }

    if (!/^[0-9]{10}$/.test(room.phone)) {
      alert("Enter a valid 10-digit phone number.");
      return;
    }

    const saved = loadRooms();
    saved.unshift(room);

    try {
      localStorage.setItem(
        "roomkhoj_rooms",
        JSON.stringify(saved)
      );
    } catch {
      alert("Unable to save listing in this browser.");
      return;
    }

    rooms = [...sampleRooms, ...saved];

    this.reset();

    const message = document.getElementById("formMessage");
    message.textContent =
      "Your demo listing has been saved in this browser!";
    message.style.display = "block";

    displayRooms(rooms);

    document.getElementById("rooms").scrollIntoView({
      behavior: "smooth"
    });
  });

// ROOM DETAILS

function findRoom(id) {
  return rooms.find(room => String(room.id) === String(id));
}

function showDetails(id) {

  const room = findRoom(id);

  if (!room) return;

  document.getElementById("modalContent").innerHTML = `
    <h2>${escapeHTML(room.title)}</h2>

    <p>
      📍 ${escapeHTML(room.area)}<br>
      💰 Rent: ₹${Number(room.rent).toLocaleString("en-IN")} / month<br>
      🏠 Type: ${escapeHTML(room.type)}<br>
      👤 Suitable for: ${escapeHTML(room.suitable)}<br>
      🛋️ ${escapeHTML(room.furnishing)}<br>
      🔐 Deposit: ₹${Number(room.deposit).toLocaleString("en-IN")}
    </p>

    <p>${escapeHTML(room.description || "No description provided.")}</p>

    <p>Owner: ${escapeHTML(room.owner)}</p>
  `;

  document.getElementById("roomModal").style.display = "flex";
}

// CONTACT OWNER

function contactOwner(id) {

  const room = findRoom(id);

  if (!room) return;

  if (room.phone === "0000000000") {
    alert("This is a sample listing. Real owner contact will be available after publishing verified listings.");
    return;
  }

  const choice = confirm(
    "Contact " + room.owner + " by phone?"
  );

  if (choice) {
    window.location.href = "tel:" + room.phone;
  }
}

  document.getElementById("roomModal").style.display = "none";
